    function needTrans(t){

    if (eqNoTransCur(t)) return false;      /* ★② 货币代码/货币对：永不翻译 */

    if (eqTotalLock(t, 'zh')) return false; /* ★EQ-TOTAL-LABEL-LOCK-20261017：合计行三语写死，不联网翻译 */

    function transOf(text, target){

       后果：越文界面取到 rec.vi 有值 ⇒ 既不去请求、也套不上真译文 ⇒ 永远显示中文。
       这里判：target='vi' 且译文与原文一字不差 ⇒ 视为【没有译文】（让它去请求真的 vi 译文）。 */
    if (target === 'vi' && String(v) === String(text)) return null;

    function requestTrans(text, target, cb){
    if (!text || !target) return;
    if (eqNoTransCur(text)) { cb && cb(null); return; }   /* ★④ 货币代码/货币对：不发翻译请求 */
    if (eqNoTransWord(text)) { cb && cb(null); return; }  /* ★EQ-WORD-NOTRANS-LOCK-20261016：公司名/人名缩写不发请求 */
    if (eqHeadRec(text)) { cb && cb(null); return; }      /* ★EQ-HEAD-LOCK-20261016：表头三语已写死，不发请求 */
    /* ★EQ-VI-TRANS-20261016：缓存里那条「假越文译文」（rec.vi === 原文）不算数 ——
       否则这里会拿它立刻回调、根本不发请求，越文界面就永远停在中文。 */

    if (_have && !(target === 'vi' && String(_have) === String(text))) { cb && cb(_have); return; }

    if (pending[key]) { cb && cb(null); return; }
    pending[key] = true;
    fetch('/api/translate?q=' + encodeURIComponent(text) + '&target=' + target)
      .then(function(r){ return r.json(); })

        cb && cb(j.text);
      })
      .catch(function(){ delete pending[key]; cb && cb(null); });
    }
